// File: src/App.js
import React, { Component } from "react";
class DateDisplay extends Component {
    render() {
        return (
            <div>
                <h2>Date Display</h2>
                <p>Today's Date: {this.props.date}</p>
            </div>
        );
    }
}
class App extends Component {
    constructor(props) {
        super(props);
        this.state = {
            date: new Date().toLocaleDateString()
        };
    }
    render() {
        return (
            <div>
                <DateDisplay date={this.state.date} />
            </div>
        );
    }
}
export default App;
