// File: src/App.js
import React, { useState } from "react";
function App() {
    const [form, setForm] = useState({
        name: "",
        email: "",
        password: ""
    });
    const handleChange = (e) => {
        setForm({
            ...form,
            [e.target.name]: e.target.value
        });
    };

    const handleSubmit = (e) => {
        e.preventDefault();
        alert(
            `Registration Successful!\nName: ${form.name}\nEmail: ${form.email}`
        );
    };

    return (
        <div>
            <h2>Registration Form</h2>
            <form onSubmit={handleSubmit}>
                <label>Name:</label>
                <br />
                <input
                    type="text"
                    name="name"
                    value={form.name}
                    onChange={handleChange}
                    required />
                <br /><br />
                <label>Email:</label>
                <br />
                <input
                    type="email"
                    name="email"
                    value={form.email}
                    onChange={handleChange}
                    required />
                <br /><br />
                <label>Password</label>
                <br />
                <input
                    type="password"
                    name="password"
                    value={form.password}
                    onChange={handleChange}
                    required />
                <br /><br />
                <button type="submit">
                    Register
                </button>
            </form>
        </div>
    );
}
export default App;
