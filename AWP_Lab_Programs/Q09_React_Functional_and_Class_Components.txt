// 9a) File: prop/src/App.js
import React from 'react';
function Student(props) {
    return (
        <div>
            <h2>Name: {props.name}</h2>
            <h2>Age: {props.age}</h2>
        </div>
    );
}
function App() {
    return (
        <div>
            <Student name="MKR" age="20" />
        </div>
    );
}
export default App;
