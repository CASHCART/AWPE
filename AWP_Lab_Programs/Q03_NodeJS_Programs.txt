// File: architecture.js
console.log("start");
setTimeout(() => {
    console.log("Timer Executed");
}, 0);
console.log("end");

// File: http.js
const http = require('http')
const server = http.createServer((req, res) => {
    res.writeHead(200, { 'Content-Type': 'text/plain' })
    res.end('Hello, this is my first Node js server')
})
const PORT = 3000
server.listen(PORT, () => {
    console.log("server running at, the PORT:" + PORT)
})

// File: write.js
const fs = require('fs')
fs.writeFile('notes.txt', 'This is AWP class\n', (err) => {
    if (err) {
        console.log("Error in file", err)
    }
    else {
        console.log("File Data written successfully")
    }
})

// File: read.js
const fs = require('fs')
fs.readFile('notes.txt', 'UTF-8', (err, data) => {
    if (err) {
        console.log("Error in File", err)
    }
    else {
        console.log(data)
    }
})

// File: append.js
const fs = require('fs')
const content = `\nAppended on: ${new Date().toLocaleString()}`
fs.appendFile('notes.txt', content, (err) => {
    if (err) {
        console.log("Error in file", err)
    } else {
        console.log("File data appended");
    }
})

// File: unlink.js
const fs = require('fs')
fs.unlink('notes.txt', (err) => {
    if (err) {
        console.log("Error deleting file:", err)
    } else {
        console.log("File deleted successfully:");
    }
})

// File: os.js
const os = require('os')
console.log("Operating System:", os.type())
console.log("Platforming:", os.platform())
console.log("Architecture:", os.arch())
console.log("OS Release:", os.release())
console.log("Hostname:", os.hostname())
console.log("Home directory:", os.homedir())
console.log("Temporary Directory:", os.tmpdir())
const uptimeInSeconds = os.uptime()
const uptimeInHours = (uptimeInSeconds / 3600).toFixed(2)
console.log(`System Uptime = ${uptimeInSeconds} seconds`)
console.log(`System Uptime: ${uptimeInHours} Hours`)

Terminal Commands

Node.js must be installed. These examples use built-in modules and need no npm packages.

```bash
mkdir nodejs-programs
cd nodejs-programs
```

Save each code section using its commented filename. Run these in order so notes.txt exists before the read, append and unlink examples:

```bash
node architecture.js
node write.js
node read.js
node append.js
node os.js
node unlink.js
```

Run the HTTP server separately, open http://localhost:3000, and use Ctrl+C to stop it:

```bash
node http.js
```
